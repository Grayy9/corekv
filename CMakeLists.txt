cmake_minimum_required(VERSION 3.16)
project(corekv LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_library(corekv src/db.cpp src/format.cpp src/wal.cpp src/sst.cpp)
target_include_directories(corekv PUBLIC include)

add_executable(corekv_cli src/cli.cpp)
target_link_libraries(corekv_cli PRIVATE corekv)
add_executable(corekv_bench src/bench.cpp)
target_link_libraries(corekv_bench PRIVATE corekv)
add_executable(db_test tests/db_test.cpp)
target_link_libraries(db_test PRIVATE corekv)
add_executable(recovery_test tests/recovery_test.cpp)
target_link_libraries(recovery_test PRIVATE corekv)
add_executable(model_test tests/model_test.cpp)
target_link_libraries(model_test PRIVATE corekv)
add_executable(path_test tests/path_test.cpp)
target_link_libraries(path_test PRIVATE corekv)
add_executable(scan_test tests/scan_test.cpp)
target_link_libraries(scan_test PRIVATE corekv)
add_executable(batch_test tests/batch_test.cpp)
target_link_libraries(batch_test PRIVATE corekv)
add_executable(batch_recovery_test tests/batch_recovery_test.cpp)
target_link_libraries(batch_recovery_test PRIVATE corekv)
add_executable(auto_compaction_test tests/auto_compaction_test.cpp)
target_link_libraries(auto_compaction_test PRIVATE corekv)
add_executable(corekv_crash_writer tests/crash_writer.cpp)
target_link_libraries(corekv_crash_writer PRIVATE corekv)

enable_testing()
add_test(NAME db_test COMMAND db_test)
add_test(NAME recovery_test COMMAND recovery_test)
add_test(NAME model_test COMMAND model_test)
add_test(NAME path_test COMMAND path_test)
add_test(NAME scan_test COMMAND scan_test)
add_test(NAME batch_test COMMAND batch_test)
add_test(NAME batch_recovery_test COMMAND batch_recovery_test)
add_test(NAME auto_compaction_test COMMAND auto_compaction_test)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME cli_test COMMAND ${CMAKE_COMMAND} -E env "COREKV_CLI=$<TARGET_FILE:corekv_cli>" ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_SOURCE_DIR}/tests -p test_cli.py -v)
  add_test(NAME crash_test COMMAND ${CMAKE_COMMAND} -E env "COREKV_CLI=$<TARGET_FILE:corekv_cli>" "COREKV_CRASH_WRITER=$<TARGET_FILE:corekv_crash_writer>" ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_SOURCE_DIR}/tests -p test_crash.py -v)
  add_test(NAME bench_test COMMAND ${CMAKE_COMMAND} -E env "COREKV_BENCH=$<TARGET_FILE:corekv_bench>" ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_SOURCE_DIR}/tests -p test_bench.py -v)
  set_tests_properties(cli_test crash_test bench_test PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
endif()
